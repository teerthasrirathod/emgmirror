// Fast passive signal capture for candidate sensor outputs on a classic ESP32.
// All pins remain inputs; this sketch never applies GPIO pullups or drives pins.
// GPIO2 is included to monitor the stable ~0.83 V analog output found in probing.

const int pins[] = {2, 34, 35, 36, 39};
const int pinCount = sizeof(pins) / sizeof(pins[0]);

int averageRaw(int pin) {
  uint32_t total = 0;
  for (int i = 0; i < 4; ++i) total += analogRead(pin);
  return total / 4;
}

void setup() {
  Serial.begin(115200);
  analogReadResolution(12);
  analogSetAttenuation(ADC_11db);
  delay(300);
  Serial.println("ms,gpio2,gpio34,gpio35,gpio36,gpio39");
}

void loop() {
  Serial.print(millis());
  for (int i = 0; i < pinCount; ++i) {
    Serial.printf(",%d", averageRaw(pins[i]));
  }
  Serial.println();
  delay(10); // nominally ~100 rows/s; ADC and serial time add a little jitter
}
